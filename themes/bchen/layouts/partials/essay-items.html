{{- /* Given a page collection, drops Substack posts (params.external, from content/blog/_content.gotmpl) whose title matches a local post, so cross-posted essays appear once. Returns the remaining pages. */ -}} {{- $seen := dict -}} {{- range . }}{{ if not .Params.external -}} {{- $seen = merge $seen (dict (.Title | plainify | lower | replaceRE `[^a-z0-9]+` "") true) -}} {{- end }}{{ end -}} {{- $dups := slice -}} {{- range . }}{{ if .Params.external -}} {{- if index $seen (.Title | plainify | lower | replaceRE `[^a-z0-9]+` "") }}{{ $dups = $dups | append .Path }}{{ end -}} {{- end }}{{ end -}} {{- return (where . "Path" "not in" $dups) -}}
